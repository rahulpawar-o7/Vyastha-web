import asyncio
from lib.db import db


async def main():
    users = await db.users.find(
        {},
        {
            "_id": 0,
            "email": 1,
            "created_at": 1,
            "updated_at": 1,
            "last_login": 1,
            "first_login_at": 1,
            "login_at": 1,
            "business_id": 1,
        }
    ).to_list(None)

    for user in users:
        print(user)


asyncio.run(main())
